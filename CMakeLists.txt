cmake_minimum_required(VERSION 3.16)
project(SoundFirst_PRO LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_executable(soundfirst_protocol_tests tests/test_ni_bridge.cpp)
add_test(NAME ni_protocol COMMAND soundfirst_protocol_tests)
if(WIN32)
  add_library(reaper_soundfirst_pro SHARED src/main.cpp src/csurf_soundfirst.cpp)
  target_include_directories(reaper_soundfirst_pro PRIVATE include)
  target_compile_definitions(reaper_soundfirst_pro PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601)
  target_link_libraries(reaper_soundfirst_pro PRIVATE setupapi hid ole32 sapi uuid winmm)
  set_target_properties(reaper_soundfirst_pro PROPERTIES PREFIX "")
  if(MSVC)
    target_compile_options(reaper_soundfirst_pro PRIVATE /utf-8)
  endif()
endif()
add_executable(soundfirst_routing_tests tests/test_routing.cpp)
add_test(NAME ni_routing_actual_branches COMMAND soundfirst_routing_tests)
add_executable(soundfirst_focus_tests tests/test_focus.cpp)
add_test(NAME ni_focus_actual_branch COMMAND soundfirst_focus_tests)
